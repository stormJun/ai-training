## 2.1 剪枝理论基础

### 2.1.1 彩票假设 (Lottery Ticket Hypothesis)

**核心思想**: 随机初始化的大型神经网络包含一个"中奖"子网络，该子网络可以单独训练达到原网络的性能

```python
# 彩票假设的数学表示
原始网络: f(x; θ)  其中 θ ∈ ℝ^n
子网络:   f(x; m ⊙ θ)  其中 m ∈ {0,1}^n 是二值掩码

目标: 找到掩码 m，使得
    acc(f(x; m ⊙ θ)) ≈ acc(f(x; θ))
    且 ||m||_0 << n  (大部分权重为 0)
```

#### 彩票假设验证实验

```python
import torch
import torch.nn as nn
import copy

def lottery_ticket_experiment(model, train_loader, prune_ratio=0.2, iterations=5):
    """
    验证彩票假设的实验
    """
    # 1. 保存初始权重
    init_weights = copy.deepcopy(model.state_dict())

    # 2. 训练原始网络
    print("训练原始网络...")
    train(model, train_loader, epochs=10)
    baseline_acc = evaluate(model)
    print(f"原始网络精度: {baseline_acc:.2f}%")

    # 3. 迭代剪枝 (Iterative Magnitude Pruning, IMP)
    results = []

    for iter in range(iterations):
        print(f"\n迭代 {iter+1}/{iterations}")

        # 3.1 识别最小权重
        all_weights = []
        for param in model.parameters():
            if param.requires_grad:
                all_weights.append(param.data.abs().view(-1))
        all_weights = torch.cat(all_weights)

        # 计算阈值
        threshold = torch.quantile(all_weights, prune_ratio * (iter + 1))

        # 3.2 生成掩码
        mask = {}
        for name, param in model.named_parameters():
            if param.requires_grad:
                mask[name] = (param.data.abs() > threshold).float()

        # 3.3 重置权重到初始值 (关键！)
        model.load_state_dict(init_weights)

        # 3.4 应用掩码
        for name, param in model.named_parameters():
            if name in mask:
                param.data *= mask[name]

        # 3.5 训练稀疏网络
        train_with_mask(model, mask, train_loader, epochs=10)
        sparse_acc = evaluate(model)

        # 统计稀疏度
        sparsity = 1.0 - sum(m.sum() for m in mask.values()) / sum(m.numel() for m in mask.values())

        print(f"  稀疏度: {sparsity*100:.1f}%")
        print(f"  精度: {sparse_acc:.2f}% (vs 基线 {baseline_acc:.2f}%)")

        results.append({
            'iteration': iter + 1,
            'sparsity': sparsity,
            'accuracy': sparse_acc
        })

    return results

# 迭代 2: 稀疏度 40%, 精度 90.5% (vs 基线 91.8%)

# 迭代 5: 稀疏度 90%, 精度 82.1% (vs 基线 91.8%)  ← 显著下降
```

### 2.1.2 剪枝的信息论解释

**Shannon 信息熵视角**:

```python
# 权重的信息量
H(W) = -Σ p(w) log p(w)

# 剪枝 = 降低信息熵
# 方法: 将权重分布从连续变为稀疏

# 示例
原始权重分布: W ~ N(0, 0.1)  → H(W) ≈ -∞ (连续分布)
剪枝后分布:   W' = {w if |w| > θ else 0}  → H(W') 更低 (大量 0)
```

**最小描述长度 (MDL) 原理**:

```
模型复杂度 = 数据拟合误差 + 模型编码长度

L(model) = L(data | model) + L(model)

剪枝优化:
  - 减少 L(model)  ← 更少参数
  - 轻微增加 L(data | model)  ← 精度略降

目标: 最小化总长度 L(model)
```

### 2.1.3 剪枝与正则化的关系

#### L0 范数与剪枝

```python
# L0 范数: 非零元素数量
||W||_0 = count(w ≠ 0)

# 剪枝目标
minimize: loss(W) + λ * ||W||_0

# 问题: L0 范数不可微

# 近似方法 1: L1 范数 (Lasso)
||W||_1 = Σ |w|  ← 凸优化，鼓励稀疏

# 近似方法 2: L2 范数 + 硬阈值
||W||_2 = Σ w²  ← 训练中使用
prune(W, threshold)  ← 训练后剪枝
```

#### 可学习阈值剪枝

```python
class LearnableThresholdPruning(nn.Module):
    """可学习的剪枝阈值"""

    def __init__(self, weight):
        super().__init__()
        self.weight = nn.Parameter(weight)
        # 可学习的阈值
        self.threshold = nn.Parameter(torch.tensor(0.01))

    def forward(self, x):
        # 软阈值函数 (可微分)
        pruned_weight = torch.sign(self.weight) * torch.relu(
            torch.abs(self.weight) - self.threshold
        )

        return F.linear(x, pruned_weight)

    def get_sparsity(self):
        return (self.weight.abs() < self.threshold).float().mean()

# 训练时自动学习最优阈值
model = LearnableThresholdPruning(weight)
optimizer = torch.optim.SGD([
    {'params': [model.weight]},
    {'params': [model.threshold], 'lr': 0.001}  # 较小学习率
])

# L1 正则化
loss = criterion(output, target) + 1e-4 * model.weight.abs().sum()
```

## 2.2 剪枝准则与算法

### 2.2.1 基于幅度的剪枝 (Magnitude-Based Pruning)

**最简单的准则**: 剪枝绝对值最小的权重

```python
def magnitude_pruning(model, sparsity=0.5):
    """
    幅度剪枝: 移除 |w| 最小的权重
    """
    # 1. 收集所有权重
    all_weights = []
    for name, param in model.named_parameters():
        if 'weight' in name and param.requires_grad:
            all_weights.append(param.data.view(-1))

    all_weights = torch.cat(all_weights)

    # 2. 计算阈值
    threshold = torch.quantile(all_weights.abs(), sparsity)

    # 3. 生成掩码
    masks = {}
    for name, param in model.named_parameters():
        if 'weight' in name and param.requires_grad:
            mask = (param.data.abs() > threshold).float()
            masks[name] = mask
            param.data *= mask  # 应用掩码

    # 统计
    total_params = sum(p.numel() for p in model.parameters())
    pruned_params = sum((m == 0).sum() for m in masks.values())
    print(f"剪枝: {pruned_params}/{total_params} = {pruned_params/total_params*100:.1f}%")

    return masks

# 问题: 全局阈值可能不均衡
# 示例: 第一层权重较大 → 全部保留
#       最后一层权重较小 → 全部剪掉 (灾难性)
```

```python
def layerwise_magnitude_pruning(model, sparsity_per_layer=0.5):
    """每层独立剪枝"""
    masks = {}

    for name, param in model.named_parameters():
        if 'weight' in name and param.requires_grad:
            # 每层独立计算阈值

    return masks
```

### 2.2.2 基于梯度的剪枝

**思想**: 权重的重要性 ∝ |梯度|

```python
def gradient_based_pruning(model, dataloader, sparsity=0.5):
    """
    基于梯度累积的剪枝
    """
    # 1. 累积梯度
    gradient_accumulation = {}

    model.train()
    for batch in dataloader:
        model.zero_grad()
        loss = compute_loss(model, batch)
        loss.backward()

        # 累积梯度的绝对值
        for name, param in model.named_parameters():
            if 'weight' in name and param.grad is not None:
                if name not in gradient_accumulation:
                    gradient_accumulation[name] = torch.zeros_like(param)
                gradient_accumulation[name] += param.grad.abs()

    # 2. 基于累积梯度剪枝
    all_gradients = torch.cat([g.view(-1) for g in gradient_accumulation.values()])
    threshold = torch.quantile(all_gradients, sparsity)

    masks = {}
    for name, grad_acc in gradient_accumulation.items():
        mask = (grad_acc > threshold).float()
        masks[name] = mask

        # 应用掩码
        param = dict(model.named_parameters())[name]
        param.data *= mask

    return masks

# 优点: 考虑了权重对损失的实际影响
# 缺点: 需要前向+反向传播，计算成本高
```

### 2.2.3 二阶方法: OBS (Optimal Brain Surgeon)

**理论**: 基于 Hessian 矩阵的最优剪枝

#### 数学推导

```
泰勒展开:
ΔL = ∇L^T Δw + 1/2 Δw^T H Δw

其中 H 是 Hessian 矩阵: H_ij = ∂²L / (∂w_i ∂w_j)

剪枝第 q 个权重 (w_q = 0):
约束: e_q^T Δw + w_q = 0

拉格朗日乘数法求解:
Δw* = -w_q / H_qq * H^{-1} e_q

损失增量:
ΔL_q = w_q² / (2 H_qq)

策略: 剪枝使 ΔL 最小的权重
```

---

#### OBS 实现

```python
import torch
from torch.autograd import grad

def compute_hessian_diagonal(model, dataloader):
    """
    计算 Hessian 对角线近似
    (完整 Hessian 太大，仅计算对角)
    """
    hessian_diag = {}

    for name, param in model.named_parameters():
        if 'weight' in name:
            hessian_diag[name] = torch.zeros_like(param)

    model.eval()
    for batch in dataloader:
        # 计算损失
        output = model(batch['input'])
        loss = F.cross_entropy(output, batch['target'])

        # 一阶梯度
        grads = grad(loss, model.parameters(), create_graph=True)

        # 二阶梯度 (对角线)
        for (name, param), g in zip(model.named_parameters(), grads):
            if 'weight' in name and g is not None:
                # ∂²L / ∂w² ≈ (∂L/∂w)²  (Fisher 信息矩阵近似)
                hessian_diag[name] += g ** 2

    # 平均
    for name in hessian_diag:
        hessian_diag[name] /= len(dataloader)

    return hessian_diag

def obs_pruning(model, hessian_diag, num_prune=100):
    """
    OBS 剪枝: 剪掉导致损失增加最小的权重
    """
    # 计算每个权重的剪枝成本
    costs = {}
    for name, param in model.named_parameters():
        if 'weight' in name:
            H_diag = hessian_diag[name] + 1e-8  # 避免除零
            # ΔL = w² / (2H)
            cost = param.data ** 2 / (2 * H_diag)
            costs[name] = cost

    # 找到成本最小的权重
    all_costs = torch.cat([c.view(-1) for c in costs.values()])
    all_indices = torch.argsort(all_costs)[:num_prune]

    # 剪枝 (简化版: 直接置零)
    pruned_count = 0
    for name, cost in costs.items():
        param = dict(model.named_parameters())[name]
        flat_cost = cost.view(-1)
        flat_param = param.data.view(-1)

        # 找到该层需要剪枝的权重
        layer_prune_indices = all_indices[all_indices < flat_cost.numel()]
        flat_param[layer_prune_indices] = 0

        pruned_count += len(layer_prune_indices)
        if pruned_count >= num_prune:
            break

    return pruned_count

# OBS 优点: 理论最优
# OBS 缺点: Hessian 计算成本高 (O(n²))
```

### 2.2.4 结构化剪枝算法

#### 通道剪枝 (Channel Pruning)

```python
def channel_pruning_by_batchnorm(model, prune_ratio=0.3):
    """
    基于 BatchNorm 的 γ 参数剪枝通道

    直觉: BN 的缩放因子 γ 小 → 该通道输出小 → 不重要
    """
    channel_importance = {}

    # 1. 收集所有 BN 层的 γ
    for name, module in model.named_modules():
        if isinstance(module, nn.BatchNorm2d):
            gamma = module.weight.data.abs()  # γ 的绝对值
            channel_importance[name] = gamma

    # 2. 全局排序
    all_gamma = torch.cat([g for g in channel_importance.values()])
    threshold = torch.quantile(all_gamma, prune_ratio)

    # 3. 剪枝通道
    pruned_channels = {}
    for name, gamma in channel_importance.items():
        # 保留 γ > threshold 的通道
        keep_mask = gamma > threshold
        pruned_channels[name] = keep_mask

        print(f"{name}: 保留 {keep_mask.sum()}/{len(keep_mask)} 通道")

    # 4. 实际移除通道 (修改网络结构)
    return rebuild_network(model, pruned_channels)

def rebuild_network(model, channel_masks):
    """
    根据通道掩码重建网络
    """
    new_model = copy.deepcopy(model)

    for name, module in new_model.named_modules():
        if isinstance(module, nn.Conv2d):
            # 找到对应的 BN 掩码
            bn_name = name.replace('conv', 'bn')
            if bn_name in channel_masks:
                mask = channel_masks[bn_name]

                # 修改输出通道
                module.out_channels = mask.sum().item()
                module.weight.data = module.weight.data[mask]
                if module.bias is not None:
                    module.bias.data = module.bias.data[mask]

        elif isinstance(module, nn.BatchNorm2d):
            if name in channel_masks:
                mask = channel_masks[name]

                module.num_features = mask.sum().item()
                module.weight.data = module.weight.data[mask]
                module.bias.data = module.bias.data[mask]
                module.running_mean = module.running_mean[mask]
                module.running_var = module.running_var[mask]

    return new_model

# 使用示例
pruned_model = channel_pruning_by_batchnorm(model, prune_ratio=0.3)

# 效果: ResNet-56 on CIFAR-10
# 原始: 0.85M 参数, 92.8% 精度
# 剪枝: 0.48M 参数, 92.1% 精度 (43% 压缩, -0.7% 精度)
```

#### 注意力头剪枝 (Attention Head Pruning)

```python
def compute_head_importance(model, dataloader):
    """
    计算 Transformer 注意力头的重要性
    """
    head_importance = {}

    # 为每个注意力头添加梯度钩子
    def save_head_grads(name):
        def hook(grad):
            if name not in head_importance:
                head_importance[name] = []
            # 累积梯度的 L1 范数
            head_importance[name].append(grad.abs().mean().item())
        return hook

    hooks = []
    for name, module in model.named_modules():
        if 'attn' in name:  # 注意力层
            # 假设注意力权重形状: [batch, num_heads, seq, seq]
            hook = module.attn_weights.register_hook(save_head_grads(name))
            hooks.append(hook)

    # 前向+反向传播
    model.train()
    for batch in dataloader:
        model.zero_grad()
        loss = compute_loss(model, batch)
        loss.backward()

    # 移除钩子
    for hook in hooks:
        hook.remove()

    # 计算平均重要性
    for name in head_importance:
        head_importance[name] = sum(head_importance[name]) / len(head_importance[name])

    return head_importance

def prune_attention_heads(model, head_importance, prune_ratio=0.2):
    """
    剪枝重要性最低的注意力头
    """
    # 排序所有头
    all_heads = []
    for layer_name, importance in head_importance.items():
        for head_idx, imp in enumerate(importance):
            all_heads.append((layer_name, head_idx, imp))

    all_heads.sort(key=lambda x: x[2])  # 按重要性排序

    # 选择要剪枝的头
    num_prune = int(len(all_heads) * prune_ratio)
    heads_to_prune = all_heads[:num_prune]

    # 创建剪枝掩码
    pruned_heads = {}
    for layer_name, head_idx, _ in heads_to_prune:
        if layer_name not in pruned_heads:
            pruned_heads[layer_name] = []
        pruned_heads[layer_name].append(head_idx)

    # 应用剪枝 (修改注意力层)
    for name, module in model.named_modules():
        if name in pruned_heads:
            module.prune_heads(pruned_heads[name])

    return model

# BERT-base 剪枝结果:
# 原始: 12 层 × 12 头 = 144 注意力头
# 剪枝 30%: 保留 100 头
# GLUE 平均分: 84.5 → 83.8 (-0.7 points)
# 推理速度: +35%
```

## 2.3 高级剪枝技术

### 2.3.1 动态剪枝 (Dynamic Pruning)

**思想**: 推理时根据输入动态选择激活哪些神经元

```python
class DynamicLinear(nn.Module):
    """动态剪枝的线性层"""

    def __init__(self, in_features, out_features, threshold=0.01):
        super().__init__()
        self.weight = nn.Parameter(torch.randn(out_features, in_features))
        self.threshold = threshold

    def forward(self, x):
        # 1. 根据输入激活值决定哪些权重参与计算
        input_magnitude = x.abs().mean(dim=0)  # [in_features]

        # 2. 动态掩码
        active_inputs = input_magnitude > self.threshold  # [in_features]

        # 3. 仅使用激活的输入
        x_sparse = x * active_inputs.float()
        weight_sparse = self.weight * active_inputs.unsqueeze(0)

        # 4. 计算输出
        output = F.linear(x_sparse, weight_sparse)

        return output

# 效果: 不同输入的稀疏度不同
# 简单输入 → 高稀疏度 (90%)
# 复杂输入 → 低稀疏度 (50%)
# 平均加速: 2-3x
```

### 2.3.2 稀疏训练 (Sparse Training)

**核心**: 从零开始训练稀疏网络，而非先训练后剪枝

#### RigL (Rigging the Lottery)

```python
class RigLTrainer:
    """RigL: 动态稀疏训练"""

    def __init__(self, model, sparsity=0.9):
        self.model = model
        self.sparsity = sparsity
        self.masks = self.initialize_masks()

    def initialize_masks(self):
        """初始化随机稀疏掩码"""
        masks = {}
        for name, param in self.model.named_parameters():
            if 'weight' in name:
                mask = (torch.rand_like(param) > self.sparsity).float()
                masks[name] = mask
        return masks

    def train_step(self, batch, optimizer):
        """单步训练"""
        # 1. 前向+反向传播
        loss = self.compute_loss(batch)
        loss.backward()

        # 2. RigL 更新: 每 N 步重新分配稀疏模式
        if self.step % 100 == 0:
            self.update_masks()

        # 3. 应用掩码并更新
        for name, param in self.model.named_parameters():
            if name in self.masks:
                param.grad *= self.masks[name]  # 仅更新激活的权重

        optimizer.step()
        optimizer.zero_grad()

        self.step += 1

    def update_masks(self):
        """
        RigL 掩码更新策略:
        1. 移除 (drop): 梯度最小的激活权重
        2. 生长 (grow): 梯度最大的非激活权重
        """
        drop_fraction = 0.3  # 移除 30% 的激活权重

        for name, param in self.model.named_parameters():
            if name in self.masks:
                mask = self.masks[name]
                grad = param.grad.abs()

                # 激活权重的梯度
                active_grads = grad * mask
                # 非激活权重的梯度
                inactive_grads = grad * (1 - mask)

                # 移除: 梯度最小的激活权重
                num_drop = int(mask.sum() * drop_fraction)
                if num_drop > 0:
                    drop_threshold = torch.topk(active_grads.view(-1), num_drop, largest=False)[0][-1]
                    mask[active_grads <= drop_threshold] = 0

                # 生长: 梯度最大的非激活权重
                num_grow = num_drop  # 保持总稀疏度不变
                if num_grow > 0:
                    grow_threshold = torch.topk(inactive_grads.view(-1), num_grow)[0][-1]
                    mask[inactive_grads >= grow_threshold] = 1

                self.masks[name] = mask

# RigL vs 传统剪枝:
# 传统: 密集训练 → 剪枝 → 微调
#       - 训练成本: 100%
#       - 最终精度: 90% 稀疏度下 -2%

# RigL: 稀疏训练 (动态掩码)
#       - 训练成本: 50% (稀疏计算)
```

### 2.3.3 渐进式剪枝 (Gradual Pruning)

```python
class GradualPruner:
    """
    渐进式剪枝: 逐步提高稀疏度

    优点: 避免剧烈剪枝导致的精度崩溃
    """

    def __init__(self, model, initial_sparsity=0.0, final_sparsity=0.9,
                 start_epoch=0, end_epoch=100):
        self.model = model
        self.initial_sparsity = initial_sparsity
        self.final_sparsity = final_sparsity
        self.start_epoch = start_epoch
        self.end_epoch = end_epoch

    def compute_sparsity(self, epoch):
        """
        计算当前 epoch 的目标稀疏度

        使用 cubic 调度:
        s(t) = s_f + (s_i - s_f) * (1 - t/T)³
        """
        if epoch < self.start_epoch:
            return self.initial_sparsity
        elif epoch >= self.end_epoch:
            return self.final_sparsity
        else:
            t = (epoch - self.start_epoch) / (self.end_epoch - self.start_epoch)
            sparsity = self.final_sparsity + (self.initial_sparsity - self.final_sparsity) * (1 - t) ** 3
            return sparsity

    def prune_at_epoch(self, epoch):
        """在指定 epoch 执行剪枝"""
        target_sparsity = self.compute_sparsity(epoch)

        # 收集所有权重
        all_weights = []
        for param in self.model.parameters():
            all_weights.append(param.data.abs().view(-1))
        all_weights = torch.cat(all_weights)

        # 计算阈值
        threshold = torch.quantile(all_weights, target_sparsity)

        # 应用掩码
        for param in self.model.parameters():
            mask = (param.data.abs() > threshold).float()
            param.data *= mask

        print(f"Epoch {epoch}: 目标稀疏度 {target_sparsity*100:.1f}%")

# 使用示例
pruner = GradualPruner(
    model,
    initial_sparsity=0.0,
    final_sparsity=0.9,
    start_epoch=10,   # 从第 10 epoch 开始剪枝
    end_epoch=100     # 到第 100 epoch 达到目标稀疏度
)

for epoch in range(150):
    train_one_epoch(model, train_loader)

    # 每 5 个 epoch 剪枝一次
    if epoch % 5 == 0:
        pruner.prune_at_epoch(epoch)

    evaluate(model, val_loader)

# 效果对比:
# 一次性剪枝 90%: 精度 85.2% (-6.3%)
# 渐进式剪枝 90%: 精度 89.8% (-1.7%)
```

```python
import torch
import torch.nn as nn
import torch.nn.utils.prune as prune
from copy import deepcopy

class PruningPipeline:
    """端到端剪枝流程"""

    def __init__(self, model, method='magnitude', sparsity=0.5):
        self.model = model
        self.method = method
        self.target_sparsity = sparsity
        self.original_model = deepcopy(model)

    def step1_baseline_evaluation(self, val_loader):
        """步骤 1: 评估基线性能"""
        print("=" * 50)
        print("步骤 1: 基线评估")
        print("=" * 50)

        baseline_acc = self.evaluate(self.original_model, val_loader)
        baseline_params = self.count_parameters(self.original_model)

        print(f"基线精度: {baseline_acc:.2f}%")
        print(f"基线参数: {baseline_params:,}")

        return baseline_acc, baseline_params

    def step2_sensitivity_analysis(self, val_loader):
        """步骤 2: 层级敏感度分析"""
        print("\n" + "=" * 50)
        print("步骤 2: 敏感度分析")
        print("=" * 50)

        sensitivity = {}
        baseline_acc = self.evaluate(self.model, val_loader)

        for name, module in self.model.named_modules():
            if isinstance(module, nn.Linear) or isinstance(module, nn.Conv2d):
                # 临时剪枝该层
                prune.l1_unstructured(module, name='weight', amount=self.target_sparsity)

                # 评估精度下降
                pruned_acc = self.evaluate(self.model, val_loader)
                sensitivity[name] = baseline_acc - pruned_acc

                # 恢复
                prune.remove(module, 'weight')

                print(f"{name}: 精度下降 {sensitivity[name]:.2f}%")

        # 按敏感度排序
        sorted_layers = sorted(sensitivity.items(), key=lambda x: x[1])

        print("\n敏感度排名 (Top 5 最敏感):")
        for name, sens in sorted_layers[-5:]:
            print(f"  {name}: {sens:.2f}%")

        return sensitivity

    def step3_layerwise_pruning(self, sensitivity):
        """步骤 3: 基于敏感度的分层剪枝"""
        print("\n" + "=" * 50)
        print("步骤 3: 分层剪枝")
        print("=" * 50)

        # 策略: 敏感层保留更多权重
        for name, module in self.model.named_modules():
            if name in sensitivity:
                # 动态调整稀疏度
                layer_sensitivity = sensitivity[name]
                if layer_sensitivity > 5.0:  # 高敏感
                    layer_sparsity = self.target_sparsity * 0.5  # 剪枝少一点
                elif layer_sensitivity > 2.0:  # 中等敏感
                    layer_sparsity = self.target_sparsity
                else:  # 低敏感
                    layer_sparsity = self.target_sparsity * 1.5  # 剪枝多一点

                layer_sparsity = min(layer_sparsity, 0.95)  # 最多剪 95%

                # 执行剪枝
                if self.method == 'magnitude':
                    prune.l1_unstructured(module, name='weight', amount=layer_sparsity)
                elif self.method == 'random':
                    prune.random_unstructured(module, name='weight', amount=layer_sparsity)

                print(f"{name}: 稀疏度 {layer_sparsity*100:.1f}%")

    def step4_fine_tune(self, train_loader, val_loader, epochs=10):
        """步骤 4: 微调"""
        print("\n" + "=" * 50)
        print("步骤 4: 微调剪枝模型")
        print("=" * 50)

        optimizer = torch.optim.SGD(self.model.parameters(), lr=0.001, momentum=0.9)
        criterion = nn.CrossEntropyLoss()

        for epoch in range(epochs):
            self.model.train()
            for batch_idx, (data, target) in enumerate(train_loader):
                optimizer.zero_grad()
                output = self.model(data)
                loss = criterion(output, target)
                loss.backward()

                # 重要: 仅更新未剪枝的权重
                # (prune 模块会自动处理)

                optimizer.step()

            # 验证
            val_acc = self.evaluate(self.model, val_loader)
            print(f"Epoch {epoch+1}/{epochs}: 验证精度 {val_acc:.2f}%")

    def step5_make_permanent(self):
        """步骤 5: 永久化剪枝"""
        print("\n" + "=" * 50)
        print("步骤 5: 永久化剪枝")
        print("=" * 50)

        for name, module in self.model.named_modules():
            if hasattr(module, 'weight_orig'):
                # 移除 reparameterization
                prune.remove(module, 'weight')

        print("剪枝永久化完成")

    def step6_export(self, save_path):
        """步骤 6: 导出模型"""
        print("\n" + "=" * 50)
        print("步骤 6: 导出模型")
        print("=" * 50)

        # 保存剪枝模型
        torch.save(self.model.state_dict(), f"{save_path}/pruned_model.pth")

        # 导出为稀疏格式 (可选)
        sparse_model = self.convert_to_sparse()
        torch.save(sparse_model, f"{save_path}/pruned_model_sparse.pth")

        print(f"模型已保存到 {save_path}")

    # 辅助方法
    def evaluate(self, model, val_loader):
        """评估精度"""
        model.eval()
        correct = 0
        total = 0

        with torch.no_grad():
            for data, target in val_loader:
                output = model(data)
                pred = output.argmax(dim=1)
                correct += pred.eq(target).sum().item()
                total += target.size(0)

        return 100.0 * correct / total

    def count_parameters(self, model):
        """统计参数量"""
        return sum(p.numel() for p in model.parameters() if p.requires_grad)

    def convert_to_sparse(self):
        """转换为稀疏格式 (COO)"""
        sparse_model = {}

        for name, param in self.model.named_parameters():
            if (param.data == 0).float().mean() > 0.5:  # 稀疏度 > 50%
                # 转换为稀疏张量
                sparse_model[name] = param.data.to_sparse()
            else:
                sparse_model[name] = param.data

        return sparse_model

# 完整流程
pipeline = PruningPipeline(model, method='magnitude', sparsity=0.7)

# 步骤 1: 基线
baseline_acc, baseline_params = pipeline.step1_baseline_evaluation(val_loader)

# 步骤 2: 敏感度分析
sensitivity = pipeline.step2_sensitivity_analysis(val_loader)

# 步骤 3: 分层剪枝
pipeline.step3_layerwise_pruning(sensitivity)

# 中间评估
pruned_acc = pipeline.evaluate(pipeline.model, val_loader)
pruned_params = pipeline.count_parameters(pipeline.model)
print(f"\n剪枝后精度: {pruned_acc:.2f}% (↓ {baseline_acc - pruned_acc:.2f}%)")
print(f"剪枝后参数: {pruned_params:,} (压缩 {baseline_params/pruned_params:.2f}x)")

# 步骤 4: 微调
pipeline.step4_fine_tune(train_loader, val_loader, epochs=10)

# 步骤 5: 永久化
pipeline.step5_make_permanent()

# 步骤 6: 导出
pipeline.step6_export("./pruned_models")

# 最终评估
final_acc = pipeline.evaluate(pipeline.model, val_loader)
print(f"\n最终精度: {final_acc:.2f}% (vs 基线 {baseline_acc:.2f}%)")
```

```python
class StructuredPruning:
    """结构化剪枝: 移除整个通道/神经元"""

    def __init__(self, model):
        self.model = model

    def prune_conv_channels(self, layer_name, num_prune):
        """
        剪枝卷积层的输出通道
        """
        # 获取层
        conv_layer = self.get_layer(layer_name)
        bn_layer = self.get_layer(layer_name.replace('conv', 'bn'))

        # 计算通道重要性 (基于 BN 的 γ)
        channel_importance = bn_layer.weight.data.abs()

        # 找到最不重要的通道
        _, prune_indices = torch.topk(channel_importance, num_prune, largest=False)

        # 创建保留掩码
        keep_mask = torch.ones(conv_layer.out_channels, dtype=torch.bool)
        keep_mask[prune_indices] = False

        # 修改卷积层
        conv_layer.out_channels = keep_mask.sum().item()
        conv_layer.weight.data = conv_layer.weight.data[keep_mask]
        if conv_layer.bias is not None:
            conv_layer.bias.data = conv_layer.bias.data[keep_mask]

        # 修改 BN 层
        bn_layer.num_features = keep_mask.sum().item()
        bn_layer.weight.data = bn_layer.weight.data[keep_mask]
        bn_layer.bias.data = bn_layer.bias.data[keep_mask]
        bn_layer.running_mean = bn_layer.running_mean[keep_mask]
        bn_layer.running_var = bn_layer.running_var[keep_mask]

        # 修改下一层的输入通道
        next_conv = self.get_next_conv(layer_name)
        if next_conv is not None:
            next_conv.in_channels = keep_mask.sum().item()
            next_conv.weight.data = next_conv.weight.data[:, keep_mask]

        print(f"剪枝 {layer_name}: {prune_indices.numel()} 通道")

    def prune_linear_neurons(self, layer_name, num_prune):
        """
        剪枝全连接层的神经元
        """
        fc_layer = self.get_layer(layer_name)

        # 计算神经元重要性 (权重的 L1 范数)
        neuron_importance = fc_layer.weight.data.abs().sum(dim=1)

        # 找到最不重要的神经元
        _, prune_indices = torch.topk(neuron_importance, num_prune, largest=False)

        # 保留掩码
        keep_mask = torch.ones(fc_layer.out_features, dtype=torch.bool)
        keep_mask[prune_indices] = False

        # 修改该层
        fc_layer.out_features = keep_mask.sum().item()
        fc_layer.weight.data = fc_layer.weight.data[keep_mask]
        if fc_layer.bias is not None:
            fc_layer.bias.data = fc_layer.bias.data[keep_mask]

        # 修改下一层
        next_fc = self.get_next_linear(layer_name)
        if next_fc is not None:
            next_fc.in_features = keep_mask.sum().item()
            next_fc.weight.data = next_fc.weight.data[:, keep_mask]

        print(f"剪枝 {layer_name}: {num_prune} 神经元")

    # 辅助方法
    def get_layer(self, name):
        """通过名称获取层"""
        for n, m in self.model.named_modules():
            if n == name:
                return m
        return None

    def get_next_conv(self, layer_name):
        """获取下一个卷积层"""
        found = False
        for name, module in self.model.named_modules():
            if found and isinstance(module, nn.Conv2d):
                return module
            if name == layer_name:
                found = True
        return None

    def get_next_linear(self, layer_name):
        """获取下一个全连接层"""
        found = False
        for name, module in self.model.named_modules():
            if found and isinstance(module, nn.Linear):
                return module
            if name == layer_name:
                found = True
        return None

# 使用示例
pruner = StructuredPruning(model)

# 剪枝各层
pruner.prune_conv_channels('layer1.conv1', num_prune=16)
pruner.prune_conv_channels('layer2.conv1', num_prune=32)
pruner.prune_linear_neurons('fc1', num_prune=256)

# 微调
fine_tune(model, train_loader, epochs=10)

# 结构化剪枝的优势:
# 1. 实际加速 (硬件友好)
# 2. 模型尺寸真实减小
# 3. 无需稀疏计算库
```
